            if (type === 'systolic') {
                if (value < 120) return {text: 'Normal', class: 'category-normal'};
                if (value < 140) return {text: 'Elevated', class: 'category-elevated'};
                return {text: 'High', class: 'category-high'};
            } else if (type === 'diastolic') {
                if (value < 80) return {text: 'Normal', class: 'category-normal'};
                if (value < 90) return {text: 'Elevated', class: 'category-elevated'};
                return {text: 'High', class: 'category-high'};
            } else { // pulse
                if (value >= 60 && value <= 100) return {text: 'Normal', class: 'category-normal'};
                if (value < 60) return {text: 'Low', class: 'category-elevated'};
                return {text: 'High', class: 'category-high'};
            }
        }

# omron_main.py:


import sys
import signal
import asyncio
from threading import Thread, Lock
import time
from datetime import datetime
from config import Config
from omron_device import OmronDevice
from google_sheets import GoogleSheetsHandler
from webserver import create_app, socketio

class OmronSystem:
    def __init__(self):
        self.device = None
        self.google_sheets = None
        self.app = None
        self.running = False
        self.current_reading = None 
        self.system_start_time = datetime.now()
        self.recent_readings = []
        self.max_readings = 50
        self.reading_lock = Lock()
        
    def initialize(self):
        print("=" * 60)
        print("OMRON M7 AUTO-RECONNECT SYSTEM ACTIVE")
        print("=" * 60)
        self.device = OmronDevice(
            mac_address=Config.OMRON_MAC_ADDRESS,
            device_name=Config.DEVICE_NAME
        )
        
        if Config.GOOGLE_SHEET_ID:
            self.google_sheets = GoogleSheetsHandler()
            self.google_sheets.initialize()
        
        # Web server initialization
        self.app = create_app(self)
        self.app.config['SYSTEM'] = self
        
    def get_status(self):
        with self.reading_lock:
            connected = self.device.connected if self.device else False
            return {
                'connected': connected,
                'device': self.device.device_name if self.device else "Disconnected",
                'last_reading': self.current_reading,
                'uptime': str(datetime.now() - self.system_start_time).split('.')[0],
                'status': 'Connected' if connected else 'Searching...'
            }

    def get_recent_readings(self, count=10):
        with self.reading_lock:
            return self.recent_readings[:count]

    def _google_sheets_sync_worker(self):
        """Save to Google Sheets when reading is received"""
        while self.running:
            try:
                time.sleep(1)
                if self.google_sheets and self.current_reading:
                    with self.reading_lock:
                        reading = self.current_reading.copy()
                    
                    # Save to Google Sheets
                    success = self.google_sheets.save_reading(reading)
                    if success:
                        print(f"✅ Sheet Updated: {reading['systolic']}/{reading['diastolic']} mmHg")
                    
            except Exception as e:
                print(f"⚠️ Sync Error: {e}")

    def _run_async_worker(self):
        """Aggressive Auto-Reconnect Loop for Omron"""
        while self.running:
            loop = asyncio.new_event_loop()
            asyncio.set_event_loop(loop)
            try:
                # Start Omron device monitoring
                loop.run_until_complete(self.device.monitor_measurements(self.handle_new_reading))
            except Exception as e:
                print(f"❌ Device loop error: {e}")
            finally:
                loop.close()
            time.sleep(1)

    def handle_new_reading(self, reading):
        """Handle new BP reading from Omron"""
        if not reading:
            return
            
        print(f"📊 NEW READING: {reading['systolic']}/{reading['diastolic']} mmHg, {reading['pulse']} BPM")
        
        with self.reading_lock:
            self.current_reading = reading
            self.recent_readings.insert(0, reading)
            if len(self.recent_readings) > self.max_readings:
                self.recent_readings.pop()
        
        # Send to web dashboard
        socketio.emit('new_reading', reading)
        socketio.emit('system_status', self.get_status())

    def start(self):
        self.running = True
        # Bluetooth Thread
        Thread(target=self._run_async_worker, daemon=True).start()
        # Sheets Sync Thread
        Thread(target=self._google_sheets_sync_worker, daemon=True).start()
        
        print(f"🚀 Dashboard: http://localhost:{Config.PORT}")
        socketio.run(self.app, host=Config.HOST, port=Config.PORT, debug=False, use_reloader=False, allow_unsafe_werkzeug=True)

    def stop(self):
        self.running = False
        sys.exit(0)

if __name__ == "__main__":
    system = OmronSystem()
    system.initialize()
    system.start()
























# omron_device.py




import asyncio
from datetime import datetime
from bleak import BleakClient, BleakScanner
import struct

class OmronDevice:
    def __init__(self, mac_address, device_name):
        self.mac_address = mac_address
        self.device_name = device_name
        self.connected = False
        self.client = None
        
        # Omron M7 Blood Pressure Service UUIDs
        self.BP_SERVICE_UUID = "00001810-0000-1000-8000-00805f9b34fb"
        self.BP_CHAR_UUID = "00002a35-0000-1000-8000-00805f9b34fb"
        
    def notification_handler(self, characteristic, data):
        """Handle incoming BP data from Omron"""
        try:
            print(f"📥 Raw data received: {data.hex()}")
            
            # Omron M7 Intelli IT data format (8+ bytes)
            if len(data) >= 8:
                # Parse according to Omron specification
                systolic = data[1]  # Byte 1: Systolic
                diastolic = data[3]  # Byte 3: Diastolic
                pulse = data[7] if len(data) > 7 else 0  # Byte 7: Pulse
                
                # Validate data (non-zero values)
                if systolic > 0 and diastolic > 0:
                    reading = {
                        'systolic': systolic,
                        'diastolic': diastolic,
                        'pulse': pulse,
                        'timestamp': datetime.now().isoformat(),
                        'time_display': datetime.now().strftime("%H:%M:%S"),
                        'date_display': datetime.now().strftime("%Y-%m-%d"),
                        'status': 'Connected',
                        'device': self.device_name,
                        'source': 'REAL'
                    }
                    
                    # Callback to main system
                    if self.callback:
                        self.callback(reading)
                else:
                    print(f"⚠️ Invalid data: Sys={systolic}, Dia={diastolic}")
            else:
                print(f"⚠️ Incomplete data: {len(data)} bytes")
                
        except Exception as e:
            print(f"❌ Data parsing error: {e}")

    async def enable_notifications(self):
        """Enable notifications for BP data"""
        try:
            if not self.client or not self.client.is_connected:
                return False
                
            # Get services
            services = self.client.services
            
            # Find BP characteristic
            bp_char = None
            for service in services:
                if "1810" in service.uuid.lower() or service.uuid.lower() == self.BP_SERVICE_UUID.lower():
                    for char in service.characteristics:
                        if "2a35" in char.uuid.lower() or char.uuid.lower() == self.BP_CHAR_UUID.lower():
                            bp_char = char
                            print(f"✅ Found BP characteristic: {char.uuid}")
                            break
                if bp_char:
                    break
            
            if bp_char:
                # Enable notifications
                await self.client.start_notify(bp_char.uuid, self.notification_handler)
                print("🔔 Notifications enabled - Ready for measurement")
                return True
            else:
                print("❌ BP characteristic not found")
                return False
                
        except Exception as e:
            print(f"❌ Notification setup error: {e}")
            return False

    async def monitor_measurements(self, callback):
        """Monitor for BP measurements continuously"""
        self.callback = callback
        
        while True:
            try:
                print("🔍 Searching for Omron device...")
                self.connected = False
                
                # Scan for device
                device = await BleakScanner.find_device_by_address(
                    self.mac_address, 
                    timeout=10.0
                )
                
                if not device:
                    print(f"❌ Device {self.mac_address} not found")
                    await asyncio.sleep(5)
                    continue
                
                print(f"✅ Found device: {device.name or 'Unknown'} at {device.address}")
                
                # Connect to device
                async with BleakClient(device, timeout=30.0) as client:
                    self.client = client
                    self.connected = True
                    print(f"🔗 Connected to Omron device")
                    
                    # Enable notifications
                    if await self.enable_notifications():
                        print("👂 Listening for measurements...")
                        print("📢 **START MEASUREMENT ON OMRON DEVICE NOW**")
                        
                        # Keep connection alive and wait for measurements
                        try:
                            while client.is_connected:
                                await asyncio.sleep(1)
                        except:
                            pass
                    
            except asyncio.TimeoutError:
                print("⏰ Connection timeout")
            except Exception as e:
                print(f"❌ Connection error: {e}")
            
            self.connected = False
            print("🔄 Reconnecting in 3 seconds...")
            await asyncio.sleep(3)
